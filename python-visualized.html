# 1. Variables: names are tags tied to values
age = 12
age = age + 1        # a brand-new 13; the tag moves over

# 2. Data types: what a value is made of
bricks = 7           # int
water = 3.14         # float
word = "hello"       # str
is_on = True         # bool
print(type(bricks), type(water), type(word), type(is_on))

# 3. Functions: little factories
def double(x):
    return x * 2

print(double(3))     # 6
print(double("ha"))  # haha

# 4. Assembly lines
def shout(word):
    return word.upper() + "!"

result = shout(double("hi"))
print(result)        # HIHI!
